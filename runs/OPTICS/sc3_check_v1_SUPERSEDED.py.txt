def step_im(w, edge, A=A_INTER):

def im_drude(w, tau=TAU1_REF):
    sig0 = E_C ** 2 * N2D * tau / MSTAR

def kk_step(pr, edge, A=A_INTER, n=4_000_001):
    """numerical K-K of A*theta(w-edge), to check the closed form rather than assume it"""
    uu = (np.arange(1, n) + 0.5) / n
    scl = 20.0 * WG
    xp = scl * uu / (1.0 - uu)
    dx = scl / (1.0 - uu) ** 2
    re_p = np.where(xp > edge, A, 0.0)
    r0 = A if pr > edge else 0.0
    f = (re_p - r0) / (xp ** 2 - pr ** 2)
    return -(2.0 * pr / math.pi) * np.trapezoid(f * dx, uu)

TAU1 = 1e-16


def two_pole(w, ratio=3.0, w1=0.5, w2=0.5):
    return w1 / (1.0 + 1j * w * TAU1) + w2 / (1.0 + 1j * w * TAU1 * ratio)


def alpha_of(s):
    return (4.0 / math.pi) * np.arctan(-s.imag / s.real)


def alpha_at_peak(fn):
    w = np.logspace(-3, 3, 400_001) / TAU1
    s = fn(w)
    k = int(np.argmax(-s.imag))
    return float(alpha_of(s)[k]), float(w[k] * TAU1)


# ---- Q8: alpha = 1 at one frequency is reachable by a two-pole sum ----
w = np.logspace(-2.5, 1.5, 400_001) / TAU1
al = alpha_of(two_pole(w)) - 1.0
sgn = np.sign(al)
roots = []
for i in range(len(w) - 1):
    if sgn[i] * sgn[i + 1] < 0:
        lo, hi = w[i], w[i + 1]
        for _ in range(80):
            mid = math.sqrt(lo * hi)
            if (alpha_of(two_pole(np.array([mid]))[0]) - 1.0) * sgn[i] > 0:
                lo = mid
            else:
                hi = mid
        roots.append(math.sqrt(lo * hi) * TAU1)
ok8 = len(roots) >= 1
verdict("Q8", "a two-pole response attains alpha = 1 exactly at a finite frequency", ok8,
        "%d root(s): w tau_1 = %s" % (len(roots), ", ".join("%.6f" % r for r in roots)))

def mixture(w, f_intra, ratio=3.0, gam_frac=0.3, tau=TAU1):
    """Drude of weight f_intra plus an interband oscillator of weight 1 - f_intra."""
    sig0 = 2.0 * tau * f_intra * W_TOTAL / math.pi
    sp = 2.0 * (1.0 - f_intra) * W_TOTAL / math.pi
    w0 = ratio / tau
    gam = gam_frac * w0
    d = 1.0 / (1.0 + 1j * w * tau)
    o = (sp * w0 ** 2) / (w0 ** 2 - w ** 2 - 1j * gam * w)
    return sig0 * d + o

        wf = abs(sp * math.pi / 2.0 / (0.1 * W_TOTAL) - 1.0)
        # S4: two probe points by quadrature
        kk_res = 0.0
        for pr in probes[1:3]:
            uu = (np.arange(1, 400_001) + 0.5) / 400_001
            scl = 4.0 * w0
            xp = scl * uu / (1.0 - uu)
            dx = scl / (1.0 - uu) ** 2
            f = (re_(xp) - re_(np.array([pr]))[0]) / (xp ** 2 - pr ** 2)
            imk = -(2.0 * pr / math.pi) * np.trapezoid(f * dx, uu)
            scale_im = max(abs(im_(np.array([p]))[0]) for p in probes)
            kk_res = max(kk_res, abs(imk - im_(np.array([pr]))[0]) / max(1e-30, scale_im))
        n_tot += 1
        worst_kk = max(worst_kk, kk_res)
        worst_w = max(worst_w, wf)
        if kk_res < 1e-5 and wf < 1e-10:
            n_ok += 1
ok12 = n_ok == n_tot
verdict("Q12", "every (w_0, Gamma) on a 6x5 grid passes S4 and S5 at one fixed S_p", ok12,
        "%d/%d pass; worst K-K residual %.2e, worst weight error %.2e" % (n_ok, n_tot, worst_kk, worst_w))
VERD.append(("Q12", ok12))
out("       => passing S1-S13 does not identify a material. The acceptance set is a 2-parameter")
out("          family (w_0, Gamma) at fixed S_p, every member of which is 'legal'.")

def tau_ratio(a, kind, n=2_000_001):
    uu = (np.arange(1, n) + 0.5) / n * (math.pi / 2.0)
    su = np.sin(uu)
    if kind == "tf":
        ker = 1.0 / (su + a) ** 2
    else:
        ker = 1.0 / (su ** 2 + a ** 2)
    return np.trapezoid(ker, uu) / np.trapezoid(su * su * ker, uu)


r_tf = tau_ratio(a_rec, "tf")
